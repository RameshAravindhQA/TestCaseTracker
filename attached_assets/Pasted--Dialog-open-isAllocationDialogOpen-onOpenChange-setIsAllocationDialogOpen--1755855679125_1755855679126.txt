 <Dialog open={isAllocationDialogOpen} onOpenChange={setIsAllocationDialogOpen}>
                          <DialogTrigger asChild>
                            <Button size="sm" className="bg-blue-600 hover:bg-blue-700">
                              <Send className="h-4 w-4 mr-2" />
                              Allocate to Project
                            </Button>
                          </DialogTrigger>
                          <DialogContent className="max-w-2xl">
                            <DialogHeader>
                              <DialogTitle>Allocate Employees to Project</DialogTitle>
                              <DialogDescription>
                                {selectedProject && projects.find(p => p.id === selectedProject) 
                                  ? `Allocating to: ${projects.find(p => p.id === selectedProject)?.projectName}`
                                  : selectedEmployees.length > 0 
                                  ? `Allocating ${selectedEmployees.length} employee(s)`
                                  : "Select project and employees to allocate"}
                              </DialogDescription>
                            </DialogHeader>
                            <div className="space-y-4">
                              <div>

                                <Select 
                                  value={selectedProject?.toString() || ""} 
                                  onValueChange={(value) => setSelectedProject(parseInt(value))}
                                >
                                  <SelectTrigger>
                                    <SelectValue placeholder="Select project..." />
                                  </SelectTrigger>
                                  <SelectContent>
                                    {projects.map((project: any) => (
                                      <SelectItem key={project.id} value={project.id.toString()}>
                                        {project.projectName}
                                      </SelectItem>
                                    ))}
                                  </SelectContent>
                                </Select>
                              </div>

                              {/* Show selected employees */}
                              {selectedEmployees.length > 0 && (
                                <div>
                                  <Label>Selected Employees ({selectedEmployees.length})</Label>
                                  <div className="flex flex-wrap gap-2 mt-2">
                                    {selectedEmployees.map(employeeId => {
                                      const employee = employeesData.find((emp: any) => emp.id === employeeId);
                                      return employee ? (
                                        <Badge key={employeeId} variant="secondary" className="flex items-center gap-1">
                                          {employee.firstName} {employee.lastName}
                                          <button
                                            onClick={() => setSelectedEmployees(prev => prev.filter(id => id !== employeeId))}

                                          >

                                          </button>
                                        </Badge>
                                      ) : null;
                                    })}
                                  </div>
                                </div>
                              )}</div>
                              <div>
                                <Label>Role</Label>
                                <Input
                                  placeholder="e.g., Frontend Developer, Project Manager"
                                  value={allocationRole}
                                  onChange={(e) => setAllocationRole(e.target.value)}
                                />
                              </div>
                              <div className="grid grid-cols-2 gap-4">
                                <div>
                                  <Label>Start Date</Label>
                                  <Input
                                    type="date"
                                    value={allocationStartDate}
                                    onChange={(e) => setAllocationStartDate(e.target.value)}
                                  />
                                </div>
                                <div>
                                  <Label>End Date</Label>
                                  <Input
                                    type="date"
                                    value={allocationEndDate}
                                    onChange={(e) => setAllocationEndDate(e.target.value)}
                                  />
                                </div>
                              </div>
                              <div className="flex justify-end gap-2">
                                <Button variant="outline" onClick={() => setIsAllocationDialogOpen(false)}>
                                  Cancel
                                </Button>
                                <Button onClick={handleAllocateEmployees}>
                                  Allocate {selectedEmployees.length} Employee(s)
                                </Button>
                              </div>
                            </div>
                          </DialogContent>
                        </Dialog>
                      </div>
                    </div>
                  </div>
                )}
